    function startVoiceInput(button) {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        button.classList.toggle("listening");
        questionEl.value = "היי שגב, מה מצב הפרויקט?";
        toastMsg("הדפדפן הזה לא תומך בזיהוי קולי. נשארים במצב דמו.");
        return;
      }
      if (recognition) {
        recognition.stop();
        recognition = null;
        button.classList.remove("listening");
        return;
      }
      recognition = new SpeechRecognition();
      recognition.lang = "he-IL";
      recognition.interimResults = false;
      recognition.maxAlternatives = 1;
      button.classList.add("listening");
      toastMsg("שגב מקשיב. תגיד: היי שגב...");
      recognition.onresult = (event) => {
        const transcript = event.results[0][0].transcript;
        questionEl.value = transcript.includes("שגב") ? transcript : "היי שגב, " + transcript;
        ask(questionEl.value);
      };
      recognition.onerror = () => toastMsg("לא הצלחתי לקלוט קול ברור. אפשר להקליד לשגב.");
      recognition.onend = () => {
        button.classList.remove("listening");
        recognition = null;
      };
      recognition.start();
    }
