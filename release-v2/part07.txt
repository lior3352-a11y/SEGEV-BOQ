    function calcMonthly(){
      const rows=monthlyData(), prev=rows.reduce((s,r)=>s+r.previous*r.price,0), cur=rows.reduce((s,r)=>s+r.current*r.price,0), extras=Number(monthlyExtras.value||0), deductions=Number(monthlyDeductions.value||0), retention=Number(monthlyRetention.value||0), vat=Number(monthlyVat.value||0);
      const gross=cur+extras-deductions, hold=gross*retention/100, beforeVat=gross-hold, vatSum=beforeVat*vat/100, payable=beforeVat+vatSum;
      monthlyPrevious.textContent=monthlyMoney(prev);monthlyCurrent.textContent=monthlyMoney(cur);monthlyCumulative.textContent=monthlyMoney(prev+cur);monthlyPayable.textContent=monthlyMoney(payable);
      monthlySummary.innerHTML=`<div class="log-item">ביצוע החודש: <strong>${monthlyMoney(cur)}</strong></div><div class="log-item">חריגים/תוספות: <strong>${monthlyMoney(extras)}</strong></div><div class="log-item">קיזוזים: <strong>${monthlyMoney(deductions)}</strong></div><div class="log-item">עיכבון: <strong>${monthlyMoney(hold)}</strong></div><div class="log-item">מע״מ: <strong>${monthlyMoney(vatSum)}</strong></div><div class="log-item">לתשלום: <strong>${monthlyMoney(payable)}</strong></div>`;
      return {prev,cur,payable};
    }
