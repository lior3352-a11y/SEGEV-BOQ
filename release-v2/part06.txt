    function ask(question) {

      learn(q);

    function speak() {
      const text = lastAnswer || answerEl.textContent;
      if (!("speechSynthesis" in window)) return toastMsg("הדפדפן לא תומך בקול");
      speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      u.lang = "he-IL";
      u.rate = 0.92;
      speechSynthesis.speak(u);
    }

    function exportBackup() {
      const payload = getBackupPayload();
      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = "segev-boq-backup-" + new Date().toISOString().slice(0, 10) + ".json";
      document.body.appendChild(link);
      link.click();
      link.remove();
      URL.revokeObjectURL(url);
      setLastSaved();
      document.getElementById("backupSummary").textContent = "נוצר קובץ גיבוי לנתוני הדמו. שמור אותו בתיקיית הגיבוי או בדיסק החיצוני.";

    function importBackup(file) {
      const reader = new FileReader();
      reader.onload = () => {
        try {
          const payload = JSON.parse(reader.result);
          if (!payload.projects || !Array.isArray(payload.projects)) throw new Error("missing projects");
          projects = payload.projects;
          suppliers = payload.suppliers || defaultSuppliers;
          clients = payload.clients || defaultClients;
          tasks = payload.tasks || defaultTasks;
          team = payload.team || defaultTeam;
          alerts = payload.alerts || defaultAlerts;
          selected = projects.find((project) => project.name === payload.selectedProjectName) || projects[0];
          localStorage.setItem("segevProProjects", JSON.stringify(projects));
          localStorage.setItem("segevProSuppliers", JSON.stringify(suppliers));
          localStorage.setItem("segevProClients", JSON.stringify(clients));
          localStorage.setItem("segevProTasks", JSON.stringify(tasks));
          localStorage.setItem("segevProTeam", JSON.stringify(team));
          localStorage.setItem("segevProAlerts", JSON.stringify(alerts));
          localStorage.setItem("segevProQuestions", JSON.stringify(payload.questions || {}));
          localStorage.setItem("segevProActions", JSON.stringify(payload.actions || []));
          lastAnswer = payload.lastAnswer || "";
          setLastSaved();
          renderProjects();
          renderSuppliers();
          renderClients();
          renderTasks();
          renderTeam();
          renderAlerts();
          renderLearning();
          renderActionLog();
          updateSelected();
          document.getElementById("backupSummary").textContent = "הגיבוי יובא בהצלחה. הנתונים שוחזרו למסך ולדפדפן.";
          toastMsg("הגיבוי שוחזר");
        } catch (error) {
          toastMsg("קובץ הגיבוי לא תקין");
        }
      };
      reader.readAsText(file);
    }
