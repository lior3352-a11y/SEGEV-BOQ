    function startVoiceInput(button) {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        button.classList.toggle("listening");
        questionEl.value = "היי שגב, מה מצב הפרויקט?";
        toastMsg("הדפדפן הזה לא תומך בזיהוי קולי. נשארים במצב דמו.");
        return;
      }
      if (recognition) {
        recognition.stop();
        recognition = null;
        button.classList.remove("listening");
        return;
      }
      recognition = new SpeechRecognition();
      recognition.lang = "he-IL";
      recognition.interimResults = false;
      recognition.maxAlternatives = 1;
      button.classList.add("listening");
      toastMsg("שגב מקשיב. תגיד: היי שגב...");
      recognition.onresult = (event) => {
        const transcript = event.results[0][0].transcript;
        questionEl.value = transcript.includes("שגב") ? transcript : "היי שגב, " + transcript;
        ask(questionEl.value);
      };
      recognition.onerror = () => toastMsg("לא הצלחתי לקלוט קול ברור. אפשר להקליד לשגב.");
      recognition.onend = () => {
        button.classList.remove("listening");
        recognition = null;
      };
      recognition.start();
    }

    function speak() {
      const text = lastAnswer || answerEl.textContent;
      if (!("speechSynthesis" in window)) return toastMsg("הדפדפן לא תומך בקול");
      speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(text);
      u.lang = "he-IL";
      u.rate = 0.92;
      speechSynthesis.speak(u);
    }

    function exportBackup() {
      const payload = getBackupPayload();
      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = "segev-boq-backup-" + new Date().toISOString().slice(0, 10) + ".json";
      document.body.appendChild(link);
      link.click();
      link.remove();
      URL.revokeObjectURL(url);
      setLastSaved();
      document.getElementById("backupSummary").textContent = "נוצר קובץ גיבוי לנתוני הדמו. שמור אותו בתיקיית הגיבוי או בדיסק החיצוני.";

    }

    function importBackup(file) {
      const reader = new FileReader();
      reader.onload = () => {
        try {
          const payload = JSON.parse(reader.result);
          if (!payload.projects || !Array.isArray(payload.projects)) throw new Error("missing projects");
          projects = payload.projects;
          suppliers = payload.suppliers || defaultSuppliers;
          clients = payload.clients || defaultClients;
          tasks = payload.tasks || defaultTasks;
          team = payload.team || defaultTeam;
          alerts = payload.alerts || defaultAlerts;
          selected = projects.find((project) => project.name === payload.selectedProjectName) || projects[0];
          cloudStorage.setItem("segevProProjects", JSON.stringify(projects));
          cloudStorage.setItem("segevProSuppliers", JSON.stringify(suppliers));
          cloudStorage.setItem("segevProClients", JSON.stringify(clients));
          cloudStorage.setItem("segevProTasks", JSON.stringify(tasks));
          cloudStorage.setItem("segevProTeam", JSON.stringify(team));
          cloudStorage.setItem("segevProAlerts", JSON.stringify(alerts));
          cloudStorage.setItem("segevProQuestions", JSON.stringify(payload.questions || {}));
          cloudStorage.setItem("segevProActions", JSON.stringify(payload.actions || []));
          lastAnswer = payload.lastAnswer || "";
          setLastSaved();
          renderProjects();
          renderSuppliers();
          renderClients();
          renderTasks();
          renderTeam();
          renderAlerts();
          renderLearning();
          renderActionLog();
          updateSelected();
          document.getElementById("backupSummary").textContent = "הגיבוי יובא בהצלחה. הנתונים שוחזרו למסך ולדפדפן.";
          toastMsg("הגיבוי שוחזר");
        } catch (error) {
          toastMsg("קובץ הגיבוי לא תקין");
        }
      };
      reader.readAsText(file);
    }
